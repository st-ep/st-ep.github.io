\cvsection{Research Experience}
\role{Graduate Researcher}{UC Berkeley and Texas Robotics at UT Austin}{2024\dash present}
\proj{FORM}{ (lead author)}{lets a robot identify how an unfamiliar soft material behaves from a single interaction, in~seconds instead of minutes, and reuse that physics model to plan shaping and pouring on a real Franka~arm.}
\proj{World models}{ (\mbox{co-first} author)}{proposed \mbox{query-conditioned} world models, which build the simplest physical model that answers each question about an action; showed that \mbox{vision-language}, \mbox{video-diffusion} and \mbox{action-conditioned} world models fail such questions when hidden physics~varies.}
\proj{Neural operators}{}{lead author of SetONet, a PDE surrogate that works with irregular, incomplete or~\mbox{point-cloud} sensor data; sole developer of its \mbox{open-source} PyTorch code. \mbox{Co-developed} probabilistic inverse neural~operators.}
\proj{Adaptive control}{}{sped up Function Encoder neural ODEs about 10\texttimes, enabling \mbox{real-time} MPPI control on changing terrain (BeamNG); brought SetONet into \mbox{meta-reinforcement} learning~(JAX).}
