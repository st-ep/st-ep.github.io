\section{Research Experience}

\entryhead{Graduate Researcher}{Aug 2024 -- Present}
{UC Berkeley; Texas Robotics, UT Austin (2024--2026)}{Berkeley, CA}
Advisor: Prof.\ Krishna Kumar

\projecthead{FORM: Robot Manipulation through Direct Material Law Identification (lead author)}
\begin{itemize}
    \item Identifies an explicit material law from a single robot interaction with one linear least-squares solve on the weak form of momentum balance, with no differentiation through simulation (2--5\,s, versus 10--25\,min for iterative baselines)
    \item Reuses the identified law, held fixed, to plan new actions and geometries in MPM: elastic rod insertion, golf putting, elastoplastic shaping, and pouring
    \item On hardware: open-loop shaping of Play-Doh, butter slime, and plasticine (72--78\% footprint IoU) and pouring glycerol to six target volumes (largest mean error 3.8\,mL); workshop version at CoRL 2026, full paper submitted to ICRA 2027
\end{itemize}

\projecthead{Physically Viable World Models (co-first author)}
\begin{itemize}
    \item Argues that world models for embodied AI must answer intervention queries by representing the physical structure behind action outcomes, not only predict future observations
\end{itemize}

\projecthead{Neural Operators for PDEs with Variable Inputs (SetONet)}
\begin{itemize}
    \item Designed the SetONet architecture, a permutation-invariant operator network for solving PDEs with variable input sampling
    \item Sole developer of the codebase (PyTorch); open-sourced under MIT license
    \item Created three point-cloud benchmarks that standard DeepONet cannot handle
\end{itemize}

\projecthead{Inverse Neural Operators for Ill-Posed Problems}
\begin{itemize}
    \item Co-developed the codebase for learning inverse operators via the basis-to-basis (B2B) framework
    \item Co-implemented INN and cINN architectures (additive, affine, and probabilistic variants) and iFNO
    \item Created the Chladni plate benchmark featuring multimodal inverse mappings
    \item Conducted noise sensitivity analysis of the developed models
\end{itemize}

\projecthead{Online Dynamics Adaptation for Robotic Control (Robot On Ice)}
\begin{itemize}
    \item Developed high-performance Function Encoder Neural ODE achieving ${\sim}10\times$ speedup
    \item Enables real-time MPPI control for autonomous navigation on varying terrains
    \item Validated in BeamNG driving simulation
\end{itemize}

\projecthead{Operator Based Meta Reinforcement Learning}
\begin{itemize}
    \item Developed JAX implementation of SetONet for experiments integrating operator learning into meta RL
\end{itemize}
